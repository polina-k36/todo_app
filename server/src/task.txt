

Можно сделать:

common/

      └── paginated-response.interface.ts

Например:

export interface PaginatedResponse<T> {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
    data: T[];
}

async getTasks(): Promise<PaginatedResponse<Task>>

async getCategories(): Promise<PaginatedResponse<Category>>s
